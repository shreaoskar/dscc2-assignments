#INCLUDE <STDIO.H>

INT MAIN() {
    INT NUM;
    INT *PTR;
    INT **PPTR;


    PRINTF("ENTER AN INTEGER: ");
    SCANF("%D", &NUM);

    PTR = &NUM;
    PPTR = &PTR;

    PRINTF("\NBEFORE MODIFICATION:\N");
    PRINTF("VALUE OF NUM = %D\N", NUM);


    PRINTF("\NENTER NEW VALUE TO MODIFY USING POINTER TO POINTER: ");
    SCANF("%D", *PPTR);

    PRINTF("\NAFTER MODIFICATION USING POINTER TO POINTER:\N");
    PRINTF("VALUE OF NUM = %D\N", NUM);

    RETURN 0;
}
